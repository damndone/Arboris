```python
WORKFLOW_OUTPUT_BINDINGS_VERSION = "workflow-output-bindings.v1"


def _validated_output_bindings(value: object) -> dict[str, dict[str, object]]:
    if value is None:
        return {}
    if not isinstance(value, Mapping):
        raise WorkflowExecutionError("workflow output_bindings must be an object")
    normalized: dict[str, dict[str, object]] = {}
    for port, binding in value.items():
        if not isinstance(port, str) or not port or not isinstance(binding, Mapping):
            raise WorkflowExecutionError("workflow output binding is invalid")
        schema_version = binding.get("schema_version")
        if not isinstance(schema_version, str) or not schema_version:
            raise WorkflowExecutionError(f"workflow output binding {port} has no schema_version")
        normalized[port] = dict(binding)
    return normalized
```

```python
@dataclass(frozen=True)
class PostEstimationAdapter:
    operation_id: str
    model_family: str
    branch_schema_version: str
    execute: Callable[..., WorkflowStepResult]
    result_schema_version: str
    oracle_kind: str


class PostEstimationRegistry:
    def __init__(self) -> None:
        self._adapters: dict[tuple[str, str], PostEstimationAdapter] = {}

    def register(self, adapter: PostEstimationAdapter) -> None:
        key = (adapter.operation_id, adapter.model_family)
        existing = self._adapters.get(key)
        if existing is not None and existing != adapter:
            raise PostEstimationAdmissionError(
                "POST_ESTIMATION_ADAPTER_DUPLICATE",
                f"post-estimation adapter {key!r} is already registered",
            )
        self._adapters[key] = adapter

    def require(self, operation_id: str, model_family: str) -> PostEstimationAdapter:
        adapter = self._adapters.get((operation_id, model_family))
        if adapter is not None:
            return adapter
        known_operation = any(key[0] == operation_id for key in self._adapters)
        code = (
            "POST_ESTIMATION_FAMILY_UNSUPPORTED"
            if known_operation
            else "POST_ESTIMATION_OPERATION_UNKNOWN"
        )
        raise PostEstimationAdmissionError(
            code,
            f"post-estimation {operation_id!r} does not support family {model_family!r}",
        )

    def declarations(self) -> tuple[PostEstimationAdapter, ...]:
        return tuple(self._adapters[key] for key in sorted(self._adapters))
```

`require` 对未知 operation 和 unsupported family 使用不同 stable reason code；不得返回 OLS default。
